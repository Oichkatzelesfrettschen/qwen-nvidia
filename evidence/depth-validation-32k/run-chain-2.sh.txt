#!/bin/sh
# Wait for the first chain, then probe the 0.8B rung: two reasoning distills and the regular checkpoint at Q4_K_M and F16.
set -u
while kill -0 1788407 2>/dev/null; do sleep 30; done
export QWEN_WEDGE_DEPTHS="8192 16384 32768"
export QWEN_WEDGE_GEOMETRIES="128:32 32:8"
export QWEN_WEDGE_CONDITIONAL_DEPTHS="8192 16384 32768"
for pair in qwen35-08b-opus-reason:candidates/qwen35-08b-opus-reason/Qwen3.5-0.8B.Q4_K_M.gguf qwen3-zero-coder-v2-08b:candidates/qwen3-zero-coder-v2-08b/Qwen3-Zro-Cdr-Reason-V2-0.8B-NEO3-EX-D_AU-Q4_K_M-imat.gguf qwen35-08b-bartowski:candidates/qwen35-08b-bartowski/Qwen_Qwen3.5-0.8B-Q4_K_M.gguf qwen35-08b-f16:Qwen3.5-0.8B-GGUF/Qwen3.5-0.8B-F16.gguf; do
  id=${pair%%:*}; rel=${pair#*:}
  printf "chain_start id=%s utc=%s\n" "$id" "$(date -u +%FT%TZ)"
  ~/qwen-laptop-setup/remote/probe-depth-wedge.sh "$HOME/models/$rel" "$HOME/qwen-depth-wedge-32k/$id" > "$id.driver.log" 2>&1
  printf "chain_stop id=%s status=%s utc=%s\n" "$id" "$?" "$(date -u +%FT%TZ)"
done
laptop-idle
