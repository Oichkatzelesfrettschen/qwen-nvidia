#!/bin/sh
# Wait for the running 4B probe, then probe each remaining checkpoint in turn.
set -u
while kill -0 1763494 2>/dev/null; do sleep 30; done
export QWEN_WEDGE_DEPTHS="8192 16384 32768"
export QWEN_WEDGE_GEOMETRIES="128:32 32:8"
export QWEN_WEDGE_CONDITIONAL_DEPTHS="8192 16384 32768"
for pair in \
  qwen38-2b-distill:Qwen3.8-2B-Distill-GGUF/Qwen3.8-2B-Q4_K_M.gguf \
  qwen35-2b:Qwen3.5-2B-GGUF/Qwen3.5-2B-Q4_K_M.gguf \
  lfm25-vl-16b:LFM2.5-VL-1.6B-GGUF/LFM2.5-VL-1.6B-Q4_K_M.gguf \
  qwen35-08b:Qwen3.5-0.8B-GGUF/Qwen3.5-0.8B-Q8_0.gguf; do
  id=${pair%%:*}; rel=${pair#*:}
  printf "chain_start id=%s utc=%s\n" "$id" "$(date -u +%FT%TZ)"
  ~/qwen-laptop-setup/remote/probe-depth-wedge.sh "$HOME/models/$rel" "$HOME/qwen-depth-wedge-32k/$id" > "$id.driver.log" 2>&1
  printf "chain_stop id=%s status=%s utc=%s\n" "$id" "$?" "$(date -u +%FT%TZ)"
done
